#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int FIFO(vector<int> pages, int frames)
{
    vector<int> memory;
    int faults = 0, pointer = 0;

    for(int page : pages)
    {
        if(find(memory.begin(), memory.end(), page) == memory.end())
        {
            faults++;

            if(memory.size() < frames)
                memory.push_back(page);
            else
            {
                memory[pointer] = page;
                pointer = (pointer + 1) % frames;
            }
        }
    }

    return faults;
}

int LRU(vector<int> pages, int frames)
{
    vector<int> memory;
    vector<int> recent;
    int faults = 0;

    for(int page : pages)
    {
        auto it = find(memory.begin(), memory.end(), page);

        if(it == memory.end())
        {
            faults++;

            if(memory.size() < frames)
            {
                memory.push_back(page);
                recent.push_back(page);
            }
            else
            {
                int lru = recent[0];

                auto pos = find(memory.begin(), memory.end(), lru);
                int index = pos - memory.begin();

                memory[index] = page;
                recent.erase(recent.begin());
                recent.push_back(page);
            }
        }
        else
        {
            recent.erase(find(recent.begin(), recent.end(), page));
            recent.push_back(page);
        }
    }

    return faults;
}

int Optimal(vector<int> pages, int frames)
{
    vector<int> memory;
    int faults = 0;

    for(int i = 0; i < pages.size(); i++)
    {
        int page = pages[i];

        if(find(memory.begin(), memory.end(), page) != memory.end())
            continue;

        faults++;

        if(memory.size() < frames)
        {
            memory.push_back(page);
        }
        else
        {
            int replaceIndex = -1;
            int farthest = -1;

            for(int j = 0; j < memory.size(); j++)
            {
                int nextUse = 999999;

                for(int k = i + 1; k < pages.size(); k++)
                {
                    if(pages[k] == memory[j])
                    {
                        nextUse = k;
                        break;
                    }
                }

                if(nextUse > farthest)
                {
                    farthest = nextUse;
                    replaceIndex = j;
                }
            }

            memory[replaceIndex] = page;
        }
    }

    return faults;
}

int main()
{
    int n, frames;

    cout << "Enter number of pages: ";
    cin >> n;

    vector<int> pages(n);

    cout << "Enter reference string:\n";

    for(int i = 0; i < n; i++)
        cin >> pages[i];

    cout << "Enter number of frames: ";
    cin >> frames;

    if(frames <= 0)
    {
        cout << "Number of frames must be positive.\n";
        return 1;
    }

    cout << "\nFIFO Page Faults: "
         << FIFO(pages, frames);

    cout << "\nLRU Page Faults: "
         << LRU(pages, frames);

    cout << "\nOptimal Page Faults: "
         << Optimal(pages, frames);

    cout << endl;

    return 0;
}
