#include <iostream>
#include <queue>
using namespace std;

int main()
{
    int n, quantum;

    cout << "Enter number of processes: ";
    cin >> n;

    int at[20], bt[20], rem[20], ct[20], tat[20], wt[20];
    bool added[20] = {false};

    for(int i = 0; i < n; i++)
    {
        cout << "Enter AT and BT for P" << i + 1 << ": ";
        cin >> at[i] >> bt[i];
        rem[i] = bt[i];
    }

    cout << "Enter Time Quantum: ";
    cin >> quantum;

    queue<int> q;
    int time = 0, completed = 0;

    while(completed < n)
    {
        for(int i = 0; i < n; i++)
        {
            if(!added[i] && at[i] <= time)
            {
                q.push(i);
                added[i] = true;
            }
        }

        if(q.empty())
        {
            time++;
            continue;
        }

        int idx = q.front();
        q.pop();

        int run = min(quantum, rem[idx]);
        time += run;
        rem[idx] -= run;

        for(int i = 0; i < n; i++)
        {
            if(!added[i] && at[i] <= time)
            {
                q.push(i);
                added[i] = true;
            }
        }

        if(rem[idx] > 0)
            q.push(idx);
        else
        {
            ct[idx] = time;
            completed++;
        }
    }

    float totalWT = 0, totalTAT = 0;

    cout << "\nP\tAT\tBT\tCT\tTAT\tWT\n";

    for(int i = 0; i < n; i++)
    {
        tat[i] = ct[i] - at[i];
        wt[i] = tat[i] - bt[i];

        totalWT += wt[i];
        totalTAT += tat[i];

        cout << "P" << i + 1 << "\t" << at[i] << "\t"
             << bt[i] << "\t" << ct[i] << "\t"
             << tat[i] << "\t" << wt[i] << endl;
    }

    cout << "\nAverage WT: " << totalWT / n;
    cout << "\nAverage TAT: " << totalTAT / n;

    return 0;
}
