#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main()
{
    int n, frames;

    cout << "Enter number of pages: ";
    cin >> n;

    vector<int> pages(n);
    cout << "Enter reference string:\n";
    for(int i = 0; i < n; i++)
        cin >> pages[i];

    cout << "Enter number of frames: ";
    cin >> frames;

    if(frames <= 0)
    {
        cout << "Number of frames must be positive.\n";
        return 1;
    }

    vector<int> memory;
    int faults = 0, pointer = 0;

    for(int page : pages)
    {
        if(find(memory.begin(), memory.end(), page) == memory.end())
        {
            faults++;

            if((int)memory.size() < frames)
                memory.push_back(page);
            else
            {
                memory[pointer] = page;
                pointer = (pointer + 1) % frames;
            }
        }
    }

    cout << "\nFIFO Page Faults: " << faults << endl;

    return 0;
}
