#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>
#include <sys/wait.h>
#include <string.h>

int main()
{
    int fd[2];
    pid_t pid;
    char message[] = "Hello from Parent Process";
    char buffer[100];

    // Create pipe
    if (pipe(fd) == -1)
    {
        perror("Pipe failed");
        return 1;
    }

    // Create child process
    pid = fork();

    if (pid < 0)
    {
        perror("Fork failed");
        return 1;
    }

    if (pid > 0)
    {
        // Parent process
        close(fd[0]);

        printf("Parent Process Created\n");
        write(fd[1], message, strlen(message) + 1);

        close(fd[1]);

        wait(NULL);
        printf("Parent Process Terminated\n");
    }
    else
    {
        // Child process
        close(fd[1]);

        read(fd[0], buffer, sizeof(buffer));

        printf("Child Process Created\n");
        printf("Message Received: %s\n", buffer);

        close(fd[0]);

        printf("Child Process Terminated\n");
        exit(0);
    }

    return 0;
}




Compile and run
gcc practical2.c -o practical2
./practical2
