#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main()
{
    int n, frames;

    cout << "Enter number of pages: ";
    cin >> n;

    vector<int> pages(n);
    cout << "Enter reference string:\n";
    for(int i = 0; i < n; i++)
        cin >> pages[i];

    cout << "Enter number of frames: ";
    cin >> frames;

    if(frames <= 0)
    {
        cout << "Number of frames must be positive.\n";
        return 1;
    }

    vector<int> memory;
    int faults = 0;

    for(int i = 0; i < n; i++)
    {
        int page = pages[i];

        if(find(memory.begin(), memory.end(), page) != memory.end())
            continue;

        faults++;

        if((int)memory.size() < frames)
        {
            memory.push_back(page);
        }
        else
        {
            int replaceIndex = -1;
            int farthest = -1;

            for(int j = 0; j < (int)memory.size(); j++)
            {
                int nextUse = 999999;

                for(int k = i + 1; k < n; k++)
                {
                    if(pages[k] == memory[j])
                    {
                        nextUse = k;
                        break;
                    }
                }

                if(nextUse > farthest)
                {
                    farthest = nextUse;
                    replaceIndex = j;
                }
            }

            memory[replaceIndex] = page;
        }
    }

    cout << "\nOptimal Page Faults: " << faults << endl;

    return 0;
}
