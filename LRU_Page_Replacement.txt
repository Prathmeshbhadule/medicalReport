#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main()
{
    int n, frames;

    cout << "Enter number of pages: ";
    cin >> n;

    vector<int> pages(n);
    cout << "Enter reference string:\n";
    for(int i = 0; i < n; i++)
        cin >> pages[i];

    cout << "Enter number of frames: ";
    cin >> frames;

    if(frames <= 0)
    {
        cout << "Number of frames must be positive.\n";
        return 1;
    }

    vector<int> memory;
    vector<int> recent;   // front = least recently used
    int faults = 0;

    for(int page : pages)
    {
        auto it = find(memory.begin(), memory.end(), page);

        if(it == memory.end())
        {
            faults++;

            if((int)memory.size() < frames)
            {
                memory.push_back(page);
                recent.push_back(page);
            }
            else
            {
                int lru = recent[0];
                auto pos = find(memory.begin(), memory.end(), lru);
                int index = pos - memory.begin();

                memory[index] = page;

                recent.erase(recent.begin());
                recent.push_back(page);
            }
        }
        else
        {
            recent.erase(find(recent.begin(), recent.end(), page));
            recent.push_back(page);
        }
    }

    cout << "\nLRU Page Faults: " << faults << endl;

    return 0;
}
